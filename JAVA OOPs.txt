Q1) Method Overloading

class Calculator {
    // Overloaded methods
    int add(int a, int b) {
        return a + b;
    }

    double add(double a, double b) {
        return a + b;
    }

    int add(int a, int b, int c) {
        return a + b + c;
    }
}

public class MethodOverloadingDemo {
    public static void main(String[] args) {
        Calculator calc = new Calculator();
        System.out.println(calc.add(5, 10));        // 15
        System.out.println(calc.add(5.5, 4.5));    // 10.0
        System.out.println(calc.add(1, 2, 3));     // 6
    }
}

Q6)  Encapsulation with Getters/Setters

class Person {
    private String name;
    private int age;

    public String getName() {
        return name;
    }
    public void setName(String name) {
        this.name = name;
    }

    public int getAge() {
        return age;
    }
    public void setAge(int age) {
        if (age > 0) {
            this.age = age;
        }
    }
}

public class EncapsulationDemo {
    public static void main(String[] args) {
        Person p = new Person();
        p.setName("John");
        p.setAge(25);
        System.out.println(p.getName() + " - " + p.getAge());
    }
}

Q11) Upcasting & Downcasting Twist

    void sound() {
        System.out.println("Animal sound");
    }
}

    void sound() {
        System.out.println("Dog barks");
    }
    void guardHouse() {
        System.out.println("Dog is guarding the house");
    }
}

public class CastingDemo {
    public static void main(String[] args) {
        Animal a = new Dog(); // Upcasting
        a.sound();

        // Downcasting to access Dog-specific method
        Dog d = (Dog) a;
        d.guardHouse();

        // Dangerous cast
        Animal a2 = new Animal();
        Dog d2 = (Dog) a2; // What happens here?
        d2.sound();
    }
}

Q24) Tightly Encapsulation

class BankAccount {
    // All instance variables are private (tight encapsulation)
    private String accountHolder;
    private double balance;

    // Constructor
    public BankAccount(String accountHolder, double balance) {
        this.accountHolder = accountHolder;
        this.balance = balance;
    }

    // Getter and Setter for accountHolder
    public String getAccountHolder() {
        return accountHolder;
    }
    public void setAccountHolder(String accountHolder) {
        this.accountHolder = accountHolder;
    }

    // Getter and Setter for balance
    public double getBalance() {
        return balance;
    }
    public void deposit(double amount) {
        if (amount > 0) {
            balance += amount;
        }
    }
    public void withdraw(double amount) {
        if (amount > 0 && amount <= balance) {
            balance -= amount;
        }
    }
}

public class TightlyEncapsulatedDemo {
    public static void main(String[] args) {
        BankAccount acc = new BankAccount("Alice", 1000);

        // Accessing through getters/setters only (no direct variable access)
        acc.deposit(500);
        acc.withdraw(200);

        System.out.println(acc.getAccountHolder() + " has balance: " + acc.getBalance());
    }
}
